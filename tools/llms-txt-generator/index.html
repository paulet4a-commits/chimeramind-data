import json
import os
from apify_client import ApifyClient

client = ApifyClient(os.environ["APIFY_TOKEN"])
run = client.actor("webdatatools/llms-txt-generator").call(run_input=json.loads("{\n    \"startUrls\": [\n        {\n            \"url\": \"https://crawlee.dev/js/docs/quick-start\"\n        }\n    ],\n    \"maxPages\": 50,\n    \"maxDepth\": 3\n}"))

for item in client.dataset(run["defaultDatasetId"]).iterate_items():
    print(item)
