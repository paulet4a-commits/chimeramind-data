import json
import os
from apify_client import ApifyClient

client = ApifyClient(os.environ["APIFY_TOKEN"])
run = client.actor("webdatatools/app-store-lookup").call(run_input=json.loads("{\n    \"mode\": \"lookup\",\n    \"apps\": [\n        \"324684580\",\n        \"com.duolingo.DuolingoMobile\",\n        \"search:meditation\"\n    ],\n    \"country\": \"us\"\n}"))

for item in client.dataset(run["defaultDatasetId"]).iterate_items():
    print(item)
