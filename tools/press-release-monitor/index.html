Call from code

Use the existing catalog input below as a starting point. Review the tool's input form on Apify before running it.

import json
import os
from apify_client import ApifyClient

client = ApifyClient(os.environ["APIFY_TOKEN"])
run = client.actor("webdatatools/press-release-monitor").call(run_input=json.loads("{\n    \"sources\": [\n        \"prnewswire\",\n        \"businesswire\",\n        \"globenewswire\"\n    ],\n    \"keywords\": [\n        \"funding\",\n        \"acquisition\",\n        \"partnership\"\n    ],\n    \"maxItemsPerSource\": 100\n}"))

for item in client.dataset(run["defaultDatasetId"]).iterate_items():
    print(item)
