import json
import os
from apify_client import ApifyClient

client = ApifyClient(os.environ["APIFY_TOKEN"])
run = client.actor("webdatatools/vinted-scraper").call(run_input=json.loads("{\n    \"searchQueries\": [\n        \"nike\"\n    ]\n}"))
