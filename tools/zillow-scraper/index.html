import json
import os
from apify_client import ApifyClient

client = ApifyClient(os.environ["APIFY_TOKEN"])
run = client.actor("webdatatools/zillow-scraper").call(run_input=json.loads("{\n    \"locations\": \"Austin, TX\",\n    \"listingType\": \"sale\",\n    \"sortBy\": \"relevance\"\n}"))

for item in client.dataset(run["defaultDatasetId"]).iterate_items():
