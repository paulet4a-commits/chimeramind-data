import json
import os
from apify_client import ApifyClient

client = ApifyClient(os.environ["APIFY_TOKEN"])
run = client.actor("webdatatools/email-validator").call(run_input=json.loads("{\n    \"emails\": [\n        \"support@apify.com\",\n        \"jane.doe@gmail.com\",\n        \"info@mailinator.com\"\n    ],\n    \"checkMx\": true,\n    \"maxConcurrency\": 10\n}"))

for item in client.dataset(run["defaultDatasetId"]).iterate_items():
    print(item)
