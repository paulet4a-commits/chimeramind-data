import json
import os
from apify_client import ApifyClient

run = client.actor("webdatatools/yc-companies-scraper").call(run_input=json.loads("{\n    \"mode\": \"companies\",\n    \"query\": \"AI agents\",\n    \"batches\": [\n        \"W25\"\n    ]\n}"))

    print(item)
