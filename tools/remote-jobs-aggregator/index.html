import json
import os
from apify_client import ApifyClient

client = ApifyClient(os.environ["APIFY_TOKEN"])
run = client.actor("webdatatools/remote-jobs-aggregator").call(run_input=json.loads("{\n    \"sources\": [\n        \"remoteok\",\n        \"weworkremotely\",\n        \"hackernews\"\n    ],\n    \"keywords\": [\n        \"python\"\n    ],\n    \"maxJobs\": 100\n}"))

for item in client.dataset(run["defaultDatasetId"]).iterate_items():
    print(item)
