import json
import os
from apify_client import ApifyClient

client = ApifyClient(os.environ["APIFY_TOKEN"])
run = client.actor("webdatatools/structured-data-extractor").call(run_input=json.loads("{\n    \"urls\": [\n        \"https://www.apify.com\",\n        \"https://www.allbirds.com/products/mens-tree-dashers\"\n    ],\n    \"includeRawJsonLd\": true,\n    \"includeOpenGraph\": true\n}"))

for item in client.dataset(run["defaultDatasetId"]).iterate_items():
