Subdomain Finder (Certificate Transparency)

Subdomain Finder enumerates every subdomain of a domain from Certificate Transparency logs (crt.sh, Cert Spotter) and resolves each one — one row per subdomain, no proxies needed.

Call from code

Use the existing catalog input below as a starting point. Review the tool's input form on Apify before running it.

import json
import os
from apify_client import ApifyClient

client = ApifyClient(os.environ["APIFY_TOKEN"])
run = client.actor("webdatatools/subdomain-finder").call(run_input=json.loads("{\n    \"domains\": [\n        \"apify.com\",\n        \"github.com\"\n    ],\n    \"resolveDns\": true,\n    \"includeWildcards\": false\n}"))

for item in client.dataset(run["defaultDatasetId"]).iterate_items():
    print(item)
import { ApifyClient } from 'apify-client';

const client = new ApifyClient({ token: process.env.APIFY_TOKEN });
const run = await client.actor('webdatatools/subdomain-finder').call({
    "domains": [
        "apify.com",
        "github.com"
    ],
    "resolveDns": true,
    "includeWildcards": false
});
const { items } = await client.dataset(run.defaultDatasetId).listItems();
console.log(items);
curl -X POST "https://api.apify.com/v2/acts/webdatatools~subdomain-finder/run-sync-get-dataset-items" \
  -H "Authorization: Bearer $APIFY_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"domains":["apify.com","github.com"],"resolveDns":true,"includeWildcards":false}'
