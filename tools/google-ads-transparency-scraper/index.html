Call from code

Use the existing catalog input below as a starting point. Review the tool's input form on Apify before running it.

import json
import os
from apify_client import ApifyClient

client = ApifyClient(os.environ["APIFY_TOKEN"])
run = client.actor("webdatatools/google-ads-transparency-scraper").call(run_input=json.loads("{\n    \"advertisers\": \"nike.com\",\n    \"region\": \"US\",\n    \"format\": \"all\"\n}"))

for item in client.dataset(run["defaultDatasetId"]).iterate_items():
    print(item)
import { ApifyClient } from 'apify-client';

const client = new ApifyClient({ token: process.env.APIFY_TOKEN });

    "region": "US",
    "format": "all"

const { items } = await client.dataset(run.defaultDatasetId).listItems();
console.log(items);
